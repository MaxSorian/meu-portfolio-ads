algoritmo "Calculadora Idade"

var
  ano_nascimento, ano_atual, idade: inteiro

inicio
  escreva ("Digite o ano do seu nascimento: ")
  leia (ano_nascimento)

  escreva ("Digite o ano atual: ")
  leia (ano_atual)

  idade <- ano_atual - ano_nascimento

  escreva ("Você tem ou vai fazer ", idade, " anos!") 

fimalgoritmo
